name: Skipped grants review

# Source copy: docs/proposed-workflows/skipped-report.yml.txt. As a text file there, nothing runs it; it takes
# effect only once copied over .github/workflows/skipped-report.yml (docs/freshness.md has the command and steps).
# The change from the earlier file: it also downloads the skipped lists and data files of the five states
# refresh-states runs, and passes them to grantmatch.skipped. Same job, same schedule.
# Proposed 2026-11-26, not in use: it also takes the skipped lists and data files of Arkansas, Maryland, Montana,
# North Dakota, Tennessee and Wyoming.

# Joins what every refresh left out (skipped-*.json.gz on the data-latest release) with the programs the site
# never shows or hides, into one CSV with a plain reason per row (grantmatch/skipped.py). The CSV goes on the
# release as skipped-grants.csv; the summary replaces the body of the one open "skip-review" issue, and a
# comment says when new ones appeared. It never changes the data. Weekly, after the page watcher.

on:
  schedule:
    - cron: "53 12 * * 5"   # Fridays
  workflow_dispatch:

permissions:
  contents: write
  issues: write
  actions: write

jobs:
  report:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install -r requirements.txt
      - name: Get the skipped lists, the state program files and last week's CSV
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          mkdir -p build/skips
          gh release download data-latest -D build/skips --clobber \
            -p 'skipped-federal.json.gz' -p 'skipped-california.json.gz' -p 'skipped-illinois.json.gz' \
            -p 'skipped-iowa.json.gz' -p 'skipped-ohio.json.gz' -p 'skipped-washington.json.gz' \
            -p 'skipped-westvirginia.json.gz' -p 'skipped-newjersey.json.gz' -p 'skipped-newyork.json.gz' \
            -p 'skipped-districtofcolumbia.json.gz' -p 'skipped-minnesota.json.gz' \
            -p 'skipped-arkansas.json.gz' -p 'skipped-maryland.json.gz' -p 'skipped-northdakota.json.gz' \
            -p 'skipped-wyoming.json.gz' -p 'skipped-tennessee.json.gz' -p 'skipped-montana.json.gz' \
            -p 'skipped-foundations.jsonl.gz' -p 'skipped-community.jsonl.gz' || true
          gh release download data-latest -D build --clobber -p florida.json.gz -p california.json.gz -p illinois.json.gz -p iowa.json.gz -p ohio.json.gz -p washington.json.gz -p westvirginia.json.gz -p newjersey.json.gz -p newyork.json.gz -p districtofcolumbia.json.gz -p minnesota.json.gz || true
          gh release download data-latest -D build --clobber -p arkansas.json.gz -p maryland.json.gz -p northdakota.json.gz -p wyoming.json.gz -p tennessee.json.gz -p montana.json.gz || true
          gh release download data-latest -D build/prev --clobber -p skipped-grants.csv || echo "No earlier report; this is the first."
          ls -la build/skips build || true
      - name: Build the report
        id: report
        run: |
          python -m grantmatch.skipped --parts build/skips --florida build/florida.json.gz \
            --california build/california.json.gz --illinois build/illinois.json.gz \
            --iowa build/iowa.json.gz --ohio build/ohio.json.gz --washington build/washington.json.gz \
            --westvirginia build/westvirginia.json.gz --newjersey build/newjersey.json.gz --newyork build/newyork.json.gz \
            --districtofcolumbia build/districtofcolumbia.json.gz --minnesota build/minnesota.json.gz \
            --arkansas build/arkansas.json.gz --maryland build/maryland.json.gz \
            --northdakota build/northdakota.json.gz --wyoming build/wyoming.json.gz \
            --tennessee build/tennessee.json.gz --montana build/montana.json.gz \
            --previous build/prev/skipped-grants.csv \
            --csv build/skipped-grants.csv --md build/skipped-report.md \
            --csv-link "https://github.com/${{ github.repository }}/releases/download/data-latest/skipped-grants.csv"
      - name: Publish the CSV
        env:
          GH_TOKEN: ${{ github.token }}
        run: gh release upload data-latest build/skipped-grants.csv --clobber
      - name: Update the review issue
        env:
          GH_TOKEN: ${{ github.token }}
          NEW: ${{ steps.report.outputs.new }}
        run: |
          gh label create skip-review --color D4C5F9 --description "Grants left out, and why" 2>/dev/null || true
          open=$(gh issue list --label skip-review --state open --json number --jq '.[0].number')
          if [ -n "$open" ]; then
            gh issue edit "$open" --body-file build/skipped-report.md
            if [ "${NEW:-0}" != "0" ]; then
              gh issue comment "$open" --body "This week's report found $NEW grants or funders left out that weren't in last week's list. They are under \"New since the last report\" above, and in the CSV with today's date under First seen."
            fi
          else
            gh issue create --title "Skipped grants: who was left out, and why" --label skip-review --body-file build/skipped-report.md
          fi
      - name: Keep this schedule from expiring
        if: always()
        env:
          GH_TOKEN: ${{ github.token }}
        run: gh api -X PUT "repos/${{ github.repository }}/actions/workflows/skipped-report.yml/enable"
